# Uygulama Planı: Spec 0061 — Rol Bazlı Öğrenci & Veri İzolasyonu ve Portal Kapsamı Standartlaştırması

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** KidsCare platformundaki tüm 20 menü ve API uç noktasında katı rol bazlı veri izolasyonunu sağlamak; Veli (PARENT) rolünün yalnızca kendi çocuklarını görmesini, Öğretmen (TEACHER) rolünün yalnızca kendi sınıf(lar)ını görmesini ve Admin rolünün tüm kreş genelini yönetmesini garantilemek.

**Architecture:**

1. Backend katmanında `StudentsController.findAll` ve `StudentsService.findAll` metoduna `PARENT` rolü eklenerek `findByParent` ile doğrudan velinin linked çocukları döndürülür.
2. Frontend katmanında tüm sayfalarda tekrarlanan ve hata üreten manuel filtreleme kodları yerine merkezi bir `useStudentScope` hook'u (`apps/admin-web/src/hooks/useStudentScope.ts`) geliştirilir.
3. `/development`, `/medication`, `/menus`, `/gallery` ve diğer kritik sayfalarda `useStudentScope` entegre edilerek "Tüm Sınıflar" ve "Tüm Öğrenciler" seçenekleri velide tamamen kaldırılır, öğretmende yalnızca kendi sınıfına kilitlenir.

**Tech Stack:** NestJS 10, Fastify/Express, Prisma ORM, React 19, TypeScript, Vitest, Jest.

## Global Constraints

- Never use `any` type in TypeScript code.
- All NestJS constructor dependencies must explicitly use `@Inject(ClassName)`.
- Multi-tenancy isolation (`tenantId`) must be strictly preserved.
- No moving cards inside cards with buttons (Tactile Single Interactive Entity Rule).
- All changes must pass `./scripts/check.ps1` with 100% green output.

```ts
  async findAll(tenantId: string, user?: CurrentUserPayload): Promise<Student[]> {
    // Veli rolü için yalnızca velinin kendi çocukları döner
    if (user?.role === 'PARENT') {
      return this.findByParent(tenantId, user.userId);
    }

    // Öğretmen rolü için sadece atandığı sınıfların öğrencileri izole edilir
    if (user?.role === 'TEACHER') {
      const classroomIds = await this.getTeacherClassroomIds(tenantId, user.userId);
      if (classroomIds.length === 0) return [];
      const rows = await this.repo.findMany(tenantId, { classroomIds });
      return rows.map((row) => Student.fromPrisma(row));
    }

    // Servis personeli için yalnızca servisine atanmış öğrenciler izole edilir
    if (user?.role === 'SERVICE_STAFF') {
      const rows = await this.repo.findMany(tenantId);
      return rows
        .map((row) => Student.fromPrisma(row))
        .filter((s) => {
          const pass = (s.passport as Record<string, unknown> | null) ?? {};
          const assigned = pass['serviceStaffId'] ?? pass['assignedServiceStaffId'];
          return assigned === user.userId;
        });
    }

    const rows = await this.repo.findMany(tenantId);
    return rows.map((row) => Student.fromPrisma(row));
  }
```

Create `apps/admin-web/src/hooks/useStudentScope.ts`:

```ts
import { useState, useEffect, useMemo } from 'react';
import { useAuth } from '../features/auth/AuthContext';
import { listStudents } from '../api/students';
import { listClassrooms } from '../api/classrooms';
import type { Student, Classroom, UserRole } from '@kidscare/shared-types';

export interface UseStudentScopeResult {
  role: UserRole;
  isStaff: boolean;
  isTeacher: boolean;
  isParent: boolean;
  loading: boolean;
  classrooms: Classroom[];
  students: Student[];
  availableClassrooms: Classroom[];
  availableStudents: Student[];
  selectedClassroomId: string;
  setSelectedClassroomId: (id: string) => void;
  selectedStudentId: string;
  setSelectedStudentId: (id: string) => void;
  showClassroomFilter: boolean;
  showStudentFilter: boolean;
  activeChild?: Student | null;
  refresh: () => Promise<void>;
}

export function useStudentScope(): UseStudentScopeResult {
  const { state } = useAuth();
  const role: UserRole = state.status === 'authenticated' ? state.user.role : 'PARENT';
  const currentUserId = state.status === 'authenticated' ? state.user.id : null;

  const isStaff = role === 'SUPER_ADMIN' || role === 'ADMIN';
  const isTeacher = role === 'TEACHER';
  const isParent = role === 'PARENT';

  const [loading, setLoading] = useState(true);
  const [classrooms, setClassrooms] = useState<Classroom[]>([]);
  const [students, setStudents] = useState<Student[]>([]);
  const [selectedClassroomId, setSelectedClassroomId] = useState<string>('');
  const [selectedStudentId, setSelectedStudentId] = useState<string>('');

  async function load() {
    setLoading(true);
    try {
      const [classList, studentList] = await Promise.all([
        isParent ? Promise.resolve([]) : listClassrooms().catch(() => []),
        listStudents().catch(() => []),
      ]);
      setClassrooms(classList);
      setStudents(studentList);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    void load();
  }, [role, currentUserId]);

  // Öğretmen için sadece kendi sınıfları
  const availableClassrooms = useMemo(() => {
    if (isStaff) return classrooms;
    if (isTeacher) {
      if (!currentUserId) return [];
      const assigned = classrooms.filter((cls) =>
        cls.teachers?.some((t) => t.teacherId === currentUserId),
      );
      if (assigned.length > 0) return assigned;
      if (classrooms.length === 1) return classrooms;
      return [];
    }
    return [];
  }, [isStaff, isTeacher, classrooms, currentUserId]);

  const teacherClassroomIds = useMemo(
    () => new Set(availableClassrooms.map((c) => c.id)),
    [availableClassrooms],
  );

  // Rol bazlı erişilebilir öğrenciler
  const availableStudents = useMemo(() => {
    if (isParent) {
      // Veli: Backend zaten sadece velinin çocuklarını döner; savunmacı kontrol:
      if (currentUserId) {
        return students.filter(
          (s) =>
            s.parentId === currentUserId ||
            (s.passport as Record<string, unknown> | null)?.secondaryParentId === currentUserId,
        );
      }
      return students;
    }
    if (isTeacher) {
      if (teacherClassroomIds.size === 0) return [];
      let pool = students.filter((s) => s.classroomId && teacherClassroomIds.has(s.classroomId));
      if (selectedClassroomId) {
        pool = pool.filter((s) => s.classroomId === selectedClassroomId);
      }
      return pool;
    }
    if (selectedClassroomId) {
      return students.filter((s) => s.classroomId === selectedClassroomId);
    }
    return students;
  }, [isParent, isTeacher, students, currentUserId, teacherClassroomIds, selectedClassroomId]);

  // Otomatik seçim senkronizasyonu
  useEffect(() => {
    if (isParent) {
      if (availableStudents.length === 1) {
        setSelectedStudentId(availableStudents[0].id);
      } else if (
        availableStudents.length > 1 &&
        (!selectedStudentId || !availableStudents.some((s) => s.id === selectedStudentId))
      ) {
        setSelectedStudentId(availableStudents[0].id);
      }
    } else if (isTeacher) {
      if (availableClassrooms.length === 1 && !selectedClassroomId) {
        setSelectedClassroomId(availableClassrooms[0].id);
      }
    }
  }, [
    isParent,
    isTeacher,
    availableStudents,
    availableClassrooms,
    selectedStudentId,
    selectedClassroomId,
  ]);

  const showClassroomFilter = !isParent && availableClassrooms.length > 0;
  const showStudentFilter = isParent ? availableStudents.length > 1 : true;
  const activeChild = isParent
    ? availableStudents.find((s) => s.id === selectedStudentId) || availableStudents[0] || null
    : null;

  return {
    role,
    isStaff,
    isTeacher,
    isParent,
    loading,
    classrooms,
    students,
    availableClassrooms,
    availableStudents,
    selectedClassroomId,
    setSelectedClassroomId,
    selectedStudentId,
    setSelectedStudentId,
    showClassroomFilter,
    showStudentFilter,
    activeChild,
    refresh: load,
  };
}
```
